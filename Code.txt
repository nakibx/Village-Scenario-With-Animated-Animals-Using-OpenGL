float dayTime=0.40f, cycleSpeed=0.0003f;
float getBrightness(){ float b=sinf(dayTime*PI); return b<0?0:b; }
float lerp(float a,float b,float t){ return a+(b-a)*t; }
void setDayNightColor(float nr,float ng,float nb,float dr,float dg,float db,float br){
    glColor3f(lerp(nr,dr,br),lerp(ng,dg,br),lerp(nb,db,br));

// SPEED SLIDER
#define SLIDER_X 980.0f
#define SLIDER_Y  22.0f
#define SLIDER_W 300.0f
#define SLIDER_H  14.0f
#define SLIDER_MIN 0.00005f
#define SLIDER_MAX 0.003f
bool sliderDragging=false;

float speedToFrac(float s){
    float lo=logf(SLIDER_MIN),hi=logf(SLIDER_MAX);
    float v=logf(s<SLIDER_MIN?SLIDER_MIN:(s>SLIDER_MAX?SLIDER_MAX:s));
    return (v-lo)/(hi-lo);
}
float fracToSpeed(float f){
    return expf(logf(SLIDER_MIN)+f*(logf(SLIDER_MAX)-logf(SLIDER_MIN)));
}
void drawSpeedSlider(){
    float frac=speedToFrac(cycleSpeed), kx=SLIDER_X+frac*SLIDER_W;

bool sliderHit(float gx,float gy){
    return gx>=SLIDER_X-7&&gx<=SLIDER_X+SLIDER_W+7&&gy>=SLIDER_Y-6&&gy<=SLIDER_Y+SLIDER_H+6;

// STARS
#define NUM_STARS 80
float starX[NUM_STARS],starY[NUM_STARS],starSize[NUM_STARS];
float starTwinkle[NUM_STARS],starTwinkleSpeed[NUM_STARS],starPhase=0;
void initStars(){
    srand(42);
    for(int i=0;i<NUM_STARS;i++){
        starX[i]=(float)(rand()%W); starY[i]=H/2+(float)(rand()%(H/2));
        starSize[i]=1.5f+(float)(rand()%30)/10.0f;
        starTwinkle[i]=(float)(rand()%628)/100.0f;
        starTwinkleSpeed[i]=0.02f+(float)(rand()%50)/1000.0f;
    }
}
void drawStars(float bright){
    float night=1-bright; if(night<0.05f)return;
    for(int i=0;i<NUM_STARS;i++){
        float tw=0.5f+0.5f*sinf(starPhase*starTwinkleSpeed[i]*10+starTwinkle[i]);
        float a=night*tw;
        glColor3f(a,a*0.97f,a*0.85f);
        float sx=starX[i],sy=starY[i],sz=starSize[i]*tw;
        glLineWidth(1.5f);
        glBegin(GL_LINES);
            glVertex2f(sx-sz,sy);glVertex2f(sx+sz,sy);
            glVertex2f(sx,sy-sz);glVertex2f(sx,sy+sz);
            glVertex2f(sx-sz*.55f,sy-sz*.55f);glVertex2f(sx+sz*.55f,sy+sz*.55f);
            glVertex2f(sx+sz*.55f,sy-sz*.55f);glVertex2f(sx-sz*.55f,sy+sz*.55f);
        glEnd();
    }
}
void drawMoon(float mx,float my,float bright){
    float n=1-bright; if(n<0.05f)return;
    glColor3f(n*0.95f,n*0.95f,n*0.80f);

    for(int i=0;i<=60;i++){float a=2*PI*i/60;glVertex2f(mx+36*cosf(a),my+36*sinf(a));}glEnd();

void drawSky(float bright){
    float sf=0;
    if(dayTime>0&&dayTime<0.5f){
        sf=(dayTime-0.15f)/0.15f;if(sf<0)sf=0;if(sf>1)sf=1;
        if(dayTime>0.30f)sf=1-(dayTime-0.30f)/0.15f;if(sf<0)sf=0;
    } else if(dayTime>=0.5f){
        float t2=dayTime-0.5f;sf=(t2-0.20f)/0.10f;
        if(sf<0)sf=0;if(sf>1)sf=1;
        if(t2>0.30f)sf=1-(t2-0.30f)/0.10f;if(sf<0)sf=0;
    }
    float tR=lerp(lerp(0.01f,0.20f,bright),0.80f,sf*bright*0.6f);
    float tG=lerp(lerp(0.01f,0.55f,bright),0.38f,sf*bright*0.5f);
    float tB=lerp(lerp(0.08f,0.92f,bright),0.10f,sf*bright*0.5f);

        glBegin(GL_LINES);
            glVertex2f(splashX,splashY);
            glVertex2f(splashX+len*cosf(splashAngle[i]),splashY+len*sinf(splashAngle[i]));
        glEnd();

void update(int val){
    dayTime+=cycleSpeed; if(dayTime>1)dayTime=0;
    starPhase+=0.016f; sunRay+=0.4f; swayPhase+=0.012f; ripplePhase+=0.045f;
    cloudX1-=0.18f;if(cloudX1<-200)cloudX1=W+200;
    cloudX2-=0.12f;if(cloudX2<-200)cloudX2=W+200;
    cloudX3-=0.22f;if(cloudX3<-200)cloudX3=W+200;
    float bright=getBrightness();
    if(bright>0.2f){
        birdX+=0.8f; if(birdX>W+100)birdX=-100;
        birdY=680+20*sinf(birdX*0.02f);
        birdWing+=wingUp?4:-4;
        if(birdWing>18)wingUp=false; if(birdWing<-18)wingUp=true;
    }
    if(duckRight){duckX+=0.5f;if(duckX>W-60)duckRight=false;}
    else         {duckX-=0.5f;if(duckX<60)duckRight=true;}
    duckBob+=duckBobDir*0.15f;
    if(duckBob>3.5f)duckBobDir=-1; if(duckBob<-3.5f)duckBobDir=1;
    ropeAngle+=girlUp?1.8f:-1.8f;
    if(ropeAngle>52)girlUp=false; if(ropeAngle<-52)girlUp=true;
    girlY=-fabsf(sinf(ropeAngle*PI/180))*38;
    for(int i=0;i<NUM_FISH;i++){
        if(fishRight[i]){fishX[i]+=fishSpeed[i];if(fishX[i]>W+30)fishX[i]=-30;}
        else            {fishX[i]-=fishSpeed[i];if(fishX[i]<-30)fishX[i]=W+30;}
    }
    catTailAngle+=catTailDir*0.025f;
    if(catTailAngle>0.8f)catTailDir=-1; if(catTailAngle<-0.8f)catTailDir=1;
    catYawnCountdown-=1;
    if(catYawnCountdown<=0&&!catYawning){catYawning=true;catYawnTimer=0;catYawnCountdown=400+rand()%300;}
    if(catYawning){catYawnTimer+=0.015f;if(catYawnTimer>=1.2f){catYawning=false;catYawnTimer=0;}}
    // Frog
    frogNextJump-=1;
    if(!frogJumping&&!frogInWater&&frogNextJump<=0){
        frogJumping=true; frogFacingRight=(rand()%2==0);
        float dir=frogFacingRight?1:-1;
        frogJumpVX=dir*(4+rand()%4); frogJumpVY=8+rand()%5;
        frogNextJump=120+rand()%180;
    }
    if(frogJumping){
        frogX+=frogJumpVX; frogY+=frogJumpVY; frogJumpVY-=0.55f;
        if(frogY<177&&frogY>73&&!frogInWater){
            frogInWater=true;frogJumping=false;frogWaterTimer=60;
            splashX=frogX;splashY=frogY;splashActive=true;splashTimer=1;
            for(int i=0;i<NUM_SPLASHES;i++){
                splashAngle[i]=PI*0.5f+(rand()%180-90)*PI/180.0f;
                splashLen[i]=15+rand()%20;
            }
        }
        if(frogY<177&&!frogInWater){frogJumping=false;frogY=195;}
        if(frogX<0)frogX=0; if(frogX>W)frogX=(float)W;
    }
    if(frogInWater){frogWaterTimer-=1;if(frogWaterTimer<=0){frogInWater=false;frogX=400+rand()%400;frogY=195;}}
    if(splashActive){splashTimer-=0.018f;if(splashTimer<=0){splashActive=false;splashTimer=0;}}
    glutPostRedisplay();
    glutTimerFunc(16,update,0);
}

void keyboard(unsigned char key,int x,int y){
    switch(key){
        case 27: exit(0);
        case 'd': dayTime=0.45f; break;
        case 'n': dayTime=0;     break;
        case 's': dayTime=0.20f; break;
        case 'H': houseLightOn=!houseLightOn; break;
        case '+': cycleSpeed*=1.4f;if(cycleSpeed>SLIDER_MAX)cycleSpeed=SLIDER_MAX; break;
        case '-': cycleSpeed/=1.4f;if(cycleSpeed<SLIDER_MIN)cycleSpeed=SLIDER_MIN; break;
    }
    glutPostRedisplay();
}
void mouse(int button,int state,int x,int y){
    float gx=(float)x,gy=(float)y;
    if(button==GLUT_LEFT_BUTTON){
        if(state==GLUT_DOWN&&sliderHit(gx,gy)){
            sliderDragging=true;
            float frac=(gx-SLIDER_X)/SLIDER_W;if(frac<0)frac=0;if(frac>1)frac=1;
            cycleSpeed=fracToSpeed(frac);
        }
        if(state==GLUT_UP)sliderDragging=false;
    }
}
void motion(int x,int y){
    if(sliderDragging){
        float frac=((float)x-SLIDER_X)/SLIDER_W;if(frac<0)frac=0;if(frac>1)frac=1;
        cycleSpeed=fracToSpeed(frac);
    }
}
void reshape(int w,int h){
    glViewport(0,0,w,h);
    glMatrixMode(GL_PROJECTION);glLoadIdentity();
    gluOrtho2D(0,W,0,H);
    glMatrixMode(GL_MODELVIEW);
}

int main(int argc,char** argv){
    srand((unsigned)time(NULL));
    glutInit(&argc,argv);
    glutInitDisplayMode(GLUT_DOUBLE|GLUT_RGB);
    glutInitWindowSize(W,H);
    glutCreateWindow("Village Scene - Day/Night Cycle v11");
    glClearColor(0,0,0,1);
    initStars(); initFish();
    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutKeyboardFunc(keyboard);
    glutMouseFunc(mouse);
    glutMotionFunc(motion);
    glutTimerFunc(16,update,0);
    glutMainLoop();
    return 0;
}
